class AIRun(models.Model):
    """تتبع كل تشغيل AI"""
    
    user = models.ForeignKey('users.User', on_delete=models.CASCADE, related_name='ai_runs')
    
    # معلومات التشغيل
    provider = models.CharField(max_length=50, verbose_name='المزود')
    model = models.CharField(max_length=100, verbose_name='النموذج')
    feature = models.CharField(max_length=50, verbose_name='الميزة')
    
    # استهلاك الرموز
    prompt_tokens = models.IntegerField(default=0)
    completion_tokens = models.IntegerField(default=0)
    total_tokens = models.IntegerField(default=0)
    
    # التكلفة
    cost_usd = models.DecimalField(max_digits=10, decimal_places=6, default=0)
    
    # الأداء
    latency_ms = models.IntegerField(default=0)
    
    # الحالة
    success = models.BooleanField(default=True)
    error_message = models.TextField(blank=True)
    
    # Metadata
    request_data = models.JSONField(default=dict, blank=True)
    response_data = models.JSONField(default=dict, blank=True)
    
    created_at = models.DateTimeField(auto_now_add=True)
    
    class Meta:
        verbose_name = 'تشغيل AI'
        verbose_name_plural = 'تشغيلات AI'
        ordering = ['-created_at']
    
    def __str__(self):
        return f"{self.user.email} - {self.provider} - {self.feature}"
```

class AIStats(models.Model):
    """إحصائيات AI اليومية لكل مستخدم"""
    
    user = models.ForeignKey('users.User', on_delete=models.CASCADE, related_name='ai_stats')
    date = models.DateField(verbose_name='التاريخ')
    
    # العدد
    total_runs = models.IntegerField(default=0)
    successful_runs = models.IntegerField(default=0)
    failed_runs = models.IntegerField(default=0)
    
    # استهلاك الرموز
    total_tokens = models.IntegerField(default=0)
    total_prompt_tokens = models.IntegerField(default=0)
    total_completion_tokens = models.IntegerField(default=0)
    
    # التكلفة
    total_cost_usd = models.DecimalField(max_digits=10, decimal_places=6, default=0)
    
    # تفصيل حسب المزود
    provider_breakdown = models.JSONField(default=dict)
    # Example: {"gemini": {"runs": 10, "tokens": 5000, "cost": 0.001}, ...}
    
    # تفصيل حسب الميزة
    feature_breakdown = models.JSONField(default=dict)
    # Example: {"lesson_plan": {"runs": 5, "tokens": 3000}, ...}
    
    class Meta:
        verbose_name = 'إحصائية AI'
        verbose_name_plural = 'إحصائيات AI'
        unique_together = ['user', 'date']
    
    def __str__(self):
        return f"{self.user.email} - {self.date}"
```

class AIPromptTemplate(models.Model):
    """قوالب الاستعلامات المتاحة"""
    
    class FeatureType(models.TextChoices):
        LESSON_PLAN = 'lesson_plan', 'خطة درس'

    feature = models.CharField(max_length=50, choices=FeatureType.choices, unique=True)
    system_prompt = models.TextField(verbose_name='الاستعلام الأساسي')
    user_template = models.TextField(verbose_name='قالب الاستعلام')
    
    # إعدادات افتراضية
    default_model = models.CharField(max_length=100, blank=True)
    default_temperature = models.FloatField(default=0.7)
    default_max_tokens = models.IntegerField(default=2000)
    
    is_active = models.BooleanField(default=True)
    version = models.IntegerField(default=1)
    
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
    class Meta:
        verbose_name = 'قالب AI'
        verbose_name_plural = 'قوالب AI'
    
    def __str__(self):
        return f"{self.get_feature_display()} v{self.version}"

```python
class AIRateLimit(models.Model):
    """حدود الاستخدام لكل مستخدم"""
    
    user = models.OneToOneField('users.User', on_delete=models.CASCADE, related_name='ai_rate_limit')
    
    # الحد اليومي
    daily_limit = models.IntegerField(default=50)
    daily_used = models.IntegerField(default=0)
    daily_reset_at = models.DateTimeField()
    
    # الحد الشهري
    monthly_limit = models.IntegerField(default=500)
    monthly_used = models.IntegerField(default=0)
    monthly_reset_at = models.DateTimeField()
    
    # الحد للنظام (معلم/طالب)
    tier = models.CharField(max_length=20, default='free')
    # free: 50/day, teacher_basic: 100/day, teacher_pro: 300/day
    
    class Meta:
        verbose_name = 'حد استخدام AI'
        verbose_name_plural = 'حدود استخدام AI'
    
    def can_make_request(self):
        from django.utils import timezone
        now = timezone.now()
        
        if now >= self.daily_reset_at:
            self.daily_used = 0
            self.daily_reset_at = now + timedelta(days=1)
            self.save()
        
        return self.daily_used < self.daily_limit
```
